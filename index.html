function safeParseJSON(str){
let inStr=false,escaped=false,out='';
for(let i=0;i<str.length;i++){
const c=str[i];
if(escaped){out+=c;escaped=false;}
else if(c==='\\'&&inStr){out+=c;escaped=true;}
else if(c==='"'){inStr=!inStr;out+=c;}
else if(inStr&&c==='\n'){out+='\\n';}
else if(inStr&&c==='\r'){out+='\\r';}
else if(inStr&&c==='\t'){out+='\\t';}
else{out+=c;}
}
return JSON.parse(out);
}

function saveCfg(){
const newYear=parseInt(document.getElementById('cfg-year')?.value)||currentYear;
S.config.startDate=document.getElementById('cfg-date').value||S.config.startDate;
S.config.weeksPerSprint=Math.max(1,Math.min(4,parseInt(document.getElementById('cfg-wpsp').value)||2));
S.config.year=newYear;
S.quarters.forEach((q,i)=>{
const lbl=document.getElementById('cfg-lbl-'+i)?.value.trim();
const sp=parseInt(document.getElementById('cfg-sp-'+i)?.value)||q.sprints;
if(lbl) q.label=lbl;
q.sprints=Math.max(1,Math.min(20,sp));
});
if(newYear!==currentYear){
currentYear=newYear;
if(!availableYears.includes(newYear)){ availableYears.push(newYear); availableYears.sort((a,b)=>a-b); }
renderYearPicker();
}
closeModal(); render();
}

closeModal(); render();
}
